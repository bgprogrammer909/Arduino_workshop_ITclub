// Automatic Night Light with LDR
// When dark → LED turns ON
// When bright → LED turns OFF

int ldrPin = A0;      // LDR connected to analog pin A0
int ledPin = 9;       // LED connected to pin 9 (PWM for brightness control)
int threshold = 20;   // Below this value 

void setup() {
  pinMode(ledPin, OUTPUT);  // 
  Serial.begin(9600);
}

void loop() {
  int ldrValue = analogRead(ldrPin);
  
  Serial.println(ldrValue);
  
  if (ldrValue >1) {
    analogWrite(ledPin, 80);   // Dark → LED ON
  } else {
    analogWrite(ledPin, 0);    // Bright → LED OFF
  }
  
}
